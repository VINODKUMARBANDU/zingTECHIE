Java Integration Guide

How to connect Java code to your LLM backend, send prompt input, and send file or screenshot input (attachments).

1. API Endpoints Used

MethodEndpointPurpose
POST/api/chat/attachmentsUpload files/screenshots and extract text/snippets
POST/api/chat-with-fallbackSend prompt to selected model with fallback support
Base URL example: http://localhost:3000
Important: Token must already be saved in your app settings, or chat returns 400 (No token saved).

2. Java Flow (Prompt + Screenshot/File)

  1. Upload screenshot/file using multipart form field name files.
  2. Read promptSnippet from attachment response.
  3. Append attachment context to your user prompt.
  4. Call chat-with-fallback with model and messages.
Step A: POST /api/chat/attachments (multipart)
Step B: Collect attachments[].promptSnippet
Step C: Build final prompt with attachment context
Step D: POST /api/chat-with-fallback (json)

4. Full Java Example

package com.wheels.automation.fleetanalytics;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import okhttp3.MediaType;
import okhttp3.MultipartBody;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.RequestBody;
import okhttp3.Response;

import java.io.File;
import java.io.IOException;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;

public class ChatWithAttachmentExample {

    private static final String BASE_URL = "http://localhost:3000";
    private static final String MODEL = "gpt-4o-mini";
    private static final MediaType JSON = MediaType.parse("application/json; charset=utf-8");
    private static final MediaType OCTET = MediaType.parse("application/octet-stream");

    public static void main(String[] args) {
        OkHttpClient client = new OkHttpClient.Builder()
                .connectTimeout(Duration.ofSeconds(20))
                .readTimeout(Duration.ofSeconds(120))
                .writeTimeout(Duration.ofSeconds(120))
                .build();

        ObjectMapper mapper = new ObjectMapper();

        String prompt = "Generate test cases for login flow";
        File attachment = new File("C:/temp/requirements.png");

        if (!attachment.exists()) {
            throw new RuntimeException("Attachment file not found: " + attachment.getAbsolutePath());
        }

        List<String> snippets = uploadAttachments(client, mapper, attachment);
        String finalPrompt = buildPrompt(prompt, snippets);
        String assistantAnswer = sendChat(client, mapper, finalPrompt);

        System.out.println("Assistant response:");
        System.out.println(assistantAnswer);
    }

    private static List<String> uploadAttachments(OkHttpClient client, ObjectMapper mapper, File file) {
        try {
            RequestBody fileBody = RequestBody.create(file, OCTET);

            MultipartBody multipartBody = new MultipartBody.Builder()
                    .setType(MultipartBody.FORM)
                    .addFormDataPart("files", file.getName(), fileBody)
                    .build();

            Request request = new Request.Builder()
                    .url(BASE_URL + "/api/chat/attachments")
                    .post(multipartBody)
                    .build();

            try (Response response = client.newCall(request).execute()) {
                String responseText = response.body() != null ? response.body().string() : "";

                if (!response.isSuccessful()) {
                    throw new RuntimeException("Attachment upload failed. HTTP " + response.code() + " - " + responseText);
                }

                JsonNode root = parseJsonOrThrow(mapper, responseText, "attachments upload");
                JsonNode attachments = root.path("attachments");

                List<String> snippets = new ArrayList<>();
                if (attachments.isArray()) {
                    for (JsonNode item : attachments) {
                        String snippet = item.path("promptSnippet").asText("");
                        if (!snippet.isBlank()) {
                            snippets.add(snippet);
                        }
                    }
                }

                return snippets;
            }
        } catch (IOException e) {
            throw new RuntimeException("I/O error while uploading attachments", e);
        }
    }

    private static String sendChat(OkHttpClient client, ObjectMapper mapper, String userPrompt) {
        try {
            String escapedPrompt = toJsonStringOrThrow(mapper, userPrompt, "chat payload");

            String payload =
                    "{"
                            + "\"model\":\"" + MODEL + "\","
                            + "\"messages\":[{\"role\":\"user\",\"content\":" + escapedPrompt + "}]"
                            + "}";

            RequestBody requestBody = RequestBody.create(payload, JSON);

            Request request = new Request.Builder()
                    .url(BASE_URL + "/api/chat-with-fallback")
                    .post(requestBody)
                    .build();

            try (Response response = client.newCall(request).execute()) {
                String responseText = response.body() != null ? response.body().string() : "";

                if (!response.isSuccessful()) {
                    throw new RuntimeException("Chat request failed. HTTP " + response.code() + " - " + responseText);
                }

                JsonNode root = parseJsonOrThrow(mapper, responseText, "chat request");
                String content = root.path("data")
                        .path("choices")
                        .path(0)
                        .path("message")
                        .path("content")
                        .asText("");

                if (content.isBlank()) {
                    return "No assistant content returned. Raw response: " + responseText;
                }

                return content;
            }
        } catch (IOException e) {
            throw new RuntimeException("I/O error while sending chat request", e);
        }
    }

    private static JsonNode parseJsonOrThrow(ObjectMapper mapper, String responseText, String context) {
        try {
            return mapper.readTree(responseText);
        } catch (IOException e) {
            throw new RuntimeException(
                    "Invalid JSON returned during " + context + ". Raw response: " + responseText,
                    e
            );
        }
    }

    private static String toJsonStringOrThrow(ObjectMapper mapper, Object value, String context) {
        try {
            return mapper.writeValueAsString(value);
        } catch (JsonProcessingException e) {
            throw new RuntimeException("Failed to serialize JSON during " + context, e);
        }
    }

    private static String buildPrompt(String prompt, List<String> snippets) {
        if (snippets == null || snippets.isEmpty()) {
            return prompt;
        }

        StringBuilder sb = new StringBuilder();
        sb.append(prompt).append("\n\n");
        sb.append("[Attachment Context]\n\n");

        for (String snippet : snippets) {
            sb.append(snippet).append("\n\n");
        }

        sb.append("[End Attachment Context]");
        return sb.toString();
    }
}

5. Prompt Input and File Input Mapping

Prompt Input

  • Java variable: userPrompt
  • JSON: messages[0].content
  • Endpoint: /api/chat-with-fallback

File or Screenshot Input

  • Java File: new File("C:/path/image.png")
  • Multipart field: files
  • Endpoint: /api/chat/attachments
